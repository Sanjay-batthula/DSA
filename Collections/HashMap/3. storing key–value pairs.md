You will be given n number of different fruits and their quantities, store them in a map DS and display them the same


```
import java.util.*;
import java.lang.*;
import java.io.*;

class Codechef {
    public static void main (String[] args) throws java.lang.Exception {
        Map<String,Integer> hm = new HashMap<>();
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        sc.nextLine(); 
        for(int i=0;i<n;i++) {
            System.out.println(hm.put(sc.next(),sc.nextInt()));
        }
    }
}
```
