# 1061. Lexicographically Smallest Equivalent String

## Union-Find with Path Compression + Lexicographical Union Rule
```
class Solution {
    private final int[] p = new int[26];

    public String smallestEquivalentString(String s1, String s2, String baseStr) {
        for (int i = 0; i < p.length; ++i) {
            p[i] = i;
        }
        for (int i = 0; i < s1.length(); ++i) {
            int x = s1.charAt(i) - 'a';
            int y = s2.charAt(i) - 'a';
            int px = find(x), py = find(y);
            if (px < py) {
                p[py] = px;
            } else {
                p[px] = py;
            }
        }
        char[] s = baseStr.toCharArray();
        for (int i = 0; i < s.length; ++i) {
            s[i] = (char) ('a' + find(s[i] - 'a'));
        }
        return String.valueOf(s);
    }

    private int find(int x) {
        if (p[x] != x) {
            p[x] = find(p[x]);
        }
        return p[x];
    }
}
```

## Graph + DFS (Connected Components approach)
```
class Solution {
    public String smallestEquivalentString(String s1, String s2, String baseStr) {
        List<Integer>[] graph = new ArrayList[26];
        for(int i = 0; i < 26; i++) graph[i] = new ArrayList<>();
        
        for(int i = 0; i < s1.length(); i++){
            int a = s1.charAt(i) - 'a';
            int b = s2.charAt(i) - 'a';
            graph[a].add(b);
            graph[b].add(a);
        }
        
        int[] smallest = new int[26];
        Arrays.fill(smallest, -1);
        
        for(int i = 0; i < 26; i++){
            if(smallest[i] == -1){
                List<Integer> comp = new ArrayList<>();
                dfs(i, graph, comp, smallest);
                
                int min = 26;
                for(int x : comp) min = Math.min(min, x);
                
                for(int x : comp) smallest[x] = min;
            }
        }
        
        StringBuilder res = new StringBuilder();
        for(char c : baseStr.toCharArray()){
            res.append((char)(smallest[c - 'a'] + 'a'));
        }
        
        return res.toString();
    }
    
    void dfs(int node, List<Integer>[] graph, List<Integer> comp, int[] smallest){
        if(smallest[node] != -1) return;
        
        smallest[node] = -2;
        comp.add(node);
        
        for(int nei : graph[node]){
            dfs(nei, graph, comp, smallest);
        }
    }
}
```
