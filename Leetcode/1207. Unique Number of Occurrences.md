## Brute Force
```
class Solution {
    public boolean uniqueOccurrences(int[] arr) {
        int n = arr.length;
        int[] counts = new int[n];
        int countIndex = 0;
        boolean[] visited = new boolean[n];
        for (int i = 0; i < n; i++) {
            if (visited[i]) continue;
            int count = 1;
            for (int j = i + 1; j < n; j++) {
                if (arr[i] == arr[j]) {
                    count++;
                    visited[j] = true;
                }
            }
            counts[countIndex++] = count;
        }
        for (int i = 0; i < countIndex; i++) {
            for (int j = i + 1; j < countIndex; j++) {
                if (counts[i] == counts[j]) return false;
            }
        }
        return true;
    }
}
```


## Frequency Counting using Hashing
```
import java.util.*;

class Solution {
    public boolean uniqueOccurrences(int[] arr) {
        Map<Integer, Integer> countMap = new HashMap<>();
        
        // Count occurrences of each number
        for (int num : arr) {
            countMap.put(num, countMap.getOrDefault(num, 0) + 1);
        }
        
        // Check if all occurrences are unique
        Set<Integer> occurrences = new HashSet<>(countMap.values());
        
        return occurrences.size() == countMap.size();
    }
}
```

## Sorting-Based Approach
```
class Solution {
    public boolean uniqueOccurrences(int[] arr) {
        Arrays.sort(arr);
        List<Integer> counts = new ArrayList<>();
        int i = 0, n = arr.length;
        while (i < n) {
            int count = 1;
            while (i + 1 < n && arr[i] == arr[i + 1]) {
                count++;
                i++;
            }
            counts.add(count);
            i++;
        }
        Collections.sort(counts);
        for (int j = 1; j < counts.size(); j++) {
            if (counts.get(j).equals(counts.get(j - 1))) return false;
        }
        return true;
    }
}
```

## Array-Based Frequency Counting
```
class Solution {
    public boolean uniqueOccurrences(int[] arr) {
        int[] freq = new int[2001];
        for (int num : arr) freq[num + 1000]++;
        boolean[] seen = new boolean[1001];
        for (int f : freq) {
            if (f > 0) {
                if (seen[f]) return false;
                seen[f] = true;
            }
        }
        return true;
    }
}
```
