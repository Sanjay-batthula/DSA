## Brute force
```
class Solution {
    public int[] getNoZeroIntegers(int n) {
        for (int a = 1; a < n; a++) {
            int b = n - a;
            if (isNoZero(a) && isNoZero(b)) {
                return new int[]{a, b};
            }
        }
        return new int[]{};
    }

    private boolean isNoZero(int x) {
        while (x > 0) {
            if (x % 10 == 0) return false;
            x /= 10;
        }
        return true;
    }
}
```

## Constructive Greedy
```
class Solution {
    public int[] getNoZeroIntegers(int n) {
        int a = 1, b = n - 1;
        while (!isNoZero(a) || !isNoZero(b)) {
            a++;
            b = n - a;
        }
        return new int[]{a, b};
    }
    private boolean isNoZero(int x) {
        while (x > 0) {
            if (x % 10 == 0) return false;
            x /= 10;
        }
        return true;
    }
}
```
