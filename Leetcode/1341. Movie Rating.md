# 1341. Movie Rating

## Top-1 Aggregate Queries using Derived Tables + UNION ALL
```sql
select results
from (
  select u.name results
  from MovieRating mr
  join users u on mr.user_id = u.user_id
  group by u.name
  order by count(mr.rating) desc, name
  limit 1 
  ) as ratings
union all
select results 
from (
  select m.title results
  from MovieRating mr2
  join Movies m on mr2.movie_id = m.movie_id 
  where DATE_FORMAT(mr2.created_at, "%Y-%m") = '2020-02'
  group by m.title
  order by avg(mr2.rating) desc, m.title 
  limit 1
) movie_ratings
```

## CTE (WITH Clause)
```sql
WITH user_rating AS (
    SELECT u.name
    FROM MovieRating mr
    JOIN Users u
    ON mr.user_id = u.user_id
    GROUP BY u.user_id, u.name
    ORDER BY COUNT(*) DESC, u.name
    LIMIT 1
),
movie_rating AS (
    SELECT m.title
    FROM MovieRating mr
    JOIN Movies m
    ON mr.movie_id = m.movie_id
    WHERE created_at BETWEEN '2020-02-01' AND '2020-02-29'
    GROUP BY m.movie_id, m.title
    ORDER BY AVG(rating) DESC, m.title
    LIMIT 1
)

SELECT name AS results
FROM user_rating

UNION ALL

SELECT title
FROM movie_rating;
```

## Window Functions
```sql
WITH user_rank AS
(
SELECT
u.name,
ROW_NUMBER() OVER(
ORDER BY COUNT(*) DESC,
u.name
) rn
FROM Users u
JOIN MovieRating mr
ON u.user_id=mr.user_id
GROUP BY u.user_id,u.name
),

movie_rank AS
(
SELECT
m.title,
ROW_NUMBER() OVER(
ORDER BY AVG(rating) DESC,
m.title
) rn
FROM Movies m
JOIN MovieRating mr
ON m.movie_id=mr.movie_id
WHERE created_at BETWEEN '2020-02-01' AND '2020-02-29'
GROUP BY m.movie_id,m.title
)

SELECT name results
FROM user_rank
WHERE rn=1

UNION ALL

SELECT title
FROM movie_rank
WHERE rn=1;
```
