## Brute Force
```
class Solution {
    public boolean checkIfExist(int[] arr) {
        int n = arr.length;
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i != j && arr[i] == 2 * arr[j]) {
                    return true;
                }
            }
        }
        return false;
    }
}
```

## HashSet Approach
```
class Solution {
    public boolean checkIfExist(int[] arr) {
        Set<Integer> s = new HashSet<>();
        for (int x : arr) {
            if (s.contains(x * 2) || ((x % 2 == 0 && s.contains(x / 2)))) {
                return true;
            }
            s.add(x);
        }
        return false;
    }
}
```

## Sorting + binary search
```
class Solution {
    public boolean checkIfExist(int[] arr) {
        java.util.Arrays.sort(arr);
        for (int i = 0; i < arr.length; i++) {
            int target = arr[i] * 2;
            int idx = java.util.Arrays.binarySearch(arr, target);
            if (idx >= 0) {
                if (idx != i) return true;
                if (idx + 1 < arr.length && arr[idx + 1] == target) return true;
                if (idx - 1 >= 0 && arr[idx - 1] == target) return true;
            }
        }
        return false;
    }
}
```

## Two Pointer Approach - Wrong Answer
```
class Solution {
    public boolean checkIfExist(int[] arr) {
        java.util.Arrays.sort(arr);
        int n = arr.length;
        int i = 0, j = 1;
        while (j < n) {
            if (i == j) j++;
            else {
                long v = (long)arr[i] * 2;
                if (arr[j] == v) return true;
                if (arr[j] < v) j++;
                else i++;
            }
        }
        return false;
    }
}

## Bitmask Frequency Approach
```
class Solution {
    public boolean checkIfExist(int[] arr) {
        int offset = 1000;
        boolean[] seen = new boolean[2001];
        for (int x : arr) {
            int d = x << 1;
            if (d >= -1000 && d <= 1000 && seen[d + offset]) return true;
            if ((x & 1) == 0) {
                int h = x >> 1;
                if (h >= -1000 && h <= 1000 && seen[h + offset]) return true;
            }
            seen[x + offset] = true;
        }
        return false;
    }
}
