# 1502. Can Make Arithmetic Progression From Sequence

## Sorting based difference check
```
class Solution {
    public boolean canMakeArithmeticProgression(int[] arr) {
        Arrays.sort(arr);
        int d = arr[1] - arr[0];
        for (int i = 2; i < arr.length; ++i) {
            if (arr[i] - arr[i - 1] != d) {
                return false;
            }
        }
        return true;
    }
}
```

## Min Max formula check - Runtime Error
```
class Solution {
    public boolean canMakeArithmeticProgression(int[] arr) {
        int min = Integer.MAX_VALUE;
        int max = Integer.MIN_VALUE;
        int n = arr.length;
        for(int x : arr){
            min = Math.min(min, x);
            max = Math.max(max, x);
        }
        if((max - min) % (n - 1) != 0) return false;
        int d = (max - min) / (n - 1);
        boolean[] seen = new boolean[n];
        for(int x : arr){
            int idx = (x - min) / d;
            if(idx < 0 || idx >= n || seen[idx] || min + idx*d != x){
                return false;
            }
            seen[idx] = true;
        }
        return true;
    }
}
```
