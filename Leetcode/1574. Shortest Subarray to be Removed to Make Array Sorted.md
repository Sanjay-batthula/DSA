# 1574. Shortest Subarray to be Removed to Make Array Sorted

## Brute Force Enumeration of Subarrays
```
class Solution {
    public int findLengthOfShortestSubarray(int[] arr) {
        int n = arr.length;
        int ans = n;
        for (int l = 0; l < n; l++) {
            for (int r = l; r < n; r++) {
                if (valid(arr, l, r)) {
                    ans = Math.min(ans, r - l + 1);
                }
            }
        }
        return ans;
    }

    private boolean valid(int[] arr, int l, int r) {
        int prev = -1;
        boolean first = true;
        for (int i = 0; i < arr.length; i++) {
            if (i >= l && i <= r) continue;
            if (!first && arr[i] < prev) return false;
            prev = arr[i];
            first = false;
        }
        return true;
    }
}

## Prefix Sorted + Suffix Sorted + Binary Search
```
class Solution {
    public int findLengthOfShortestSubarray(int[] arr) {
        int n = arr.length;
        int i = 0, j = n - 1;
        while (i + 1 < n && arr[i] <= arr[i + 1]) {
            ++i;
        }
        while (j - 1 >= 0 && arr[j - 1] <= arr[j]) {
            --j;
        }
        if (i >= j) {
            return 0;
        }
        int ans = Math.min(n - i - 1, j);
        for (int l = 0; l <= i; ++l) {
            int r = search(arr, arr[l], j);
            ans = Math.min(ans, r - l - 1);
        }
        return ans;
    }
    private int search(int[] arr, int x, int left) {
        int right = arr.length;
        while (left < right) {
            int mid = (left + right) >> 1;
            if (arr[mid] >= x) {
                right = mid;
            } else {
                left = mid + 1;
            }
        }
        return left;
    }
}
```

## Two Pointers + Prefix-Suffix Merge
```
class Solution {
    public int findLengthOfShortestSubarray(int[] arr) {
        int n = arr.length;
        int left = 0;
        while (left + 1 < n && arr[left] <= arr[left + 1])
            left++;
        if (left == n - 1)
            return 0;
        int right = n - 1;
        while (arr[right - 1] <= arr[right])
            right--;
        int ans = Math.min(n - left - 1, right);
        int i = 0, j = right;
        while (i <= left && j < n) {
            if (arr[i] <= arr[j]) {
                ans = Math.min(ans, j - i - 1);
                i++;
            } else {
                j++;
            }
        }        
        return ans;
    }
}
