## 3Sum Closest Problem

# Brute Force
```java
class Solution {
    public int threeSumClosest(int[] nums, int target) {
        int n = nums.length;
        int closest = Integer.MAX_VALUE / 2;
        for (int i = 0; i < n - 2; i++) {
            for (int j = i + 1; j < n - 1; j++) {
                for (int k = j + 1; k < n; k++) {
                    int sum = nums[i] + nums[j] + nums[k];
                    if (Math.abs(sum - target) < Math.abs(closest - target)) {
                        closest = sum;
                    }
                }
            }
        }
        return closest;
    }
}
```

## Two-Pointer Approach
```
class Solution {
    public int threeSumClosest(int[] nums, int target) {
        Arrays.sort(nums);
        int ans = 1 << 30;
        int n = nums.length;
        for (int i = 0; i < n; ++i) {
            int j = i + 1, k = n - 1;
            while (j < k) {
                int t = nums[i] + nums[j] + nums[k];
                if (t == target) {
                    return t;
                }
                if (Math.abs(t - target) < Math.abs(ans - target)) {
                    ans = t;
                }
                if (t > target) {
                    --k;
                } else {
                    ++j;
                }
            }
        }
        return ans;
    }
}
```

## Binary Search Optimization
```
class Solution {
    public int threeSumClosest(int[] nums, int target) {
        Arrays.sort(nums);
        int closest = Integer.MAX_VALUE / 2;
        int n = nums.length;

        for (int i = 0; i < n - 2; i++) {
            for (int j = i + 1; j < n - 1; j++) {
                int need = target - nums[i] - nums[j];
                int idx = Arrays.binarySearch(nums, j + 1, n, need);

                // If found exact match
                if (idx >= 0) return target;

                // Otherwise check neighbors of insertion point
                idx = -idx - 1;
                if (idx < n && Math.abs(nums[i] + nums[j] + nums[idx] - target) < Math.abs(closest - target))
                    closest = nums[i] + nums[j] + nums[idx];
                if (idx - 1 > j && Math.abs(nums[i] + nums[j] + nums[idx - 1] - target) < Math.abs(closest - target))
                    closest = nums[i] + nums[j] + nums[idx - 1];
            }
        }
        return closest;
    }
}
```
