# 1633. Percentage of Users Attended a Contest

## GROUP BY Aggregation + Scalar Subquery
```
SELECT
    r.contest_id,
    ROUND(COUNT(DISTINCT r.user_id) * 100.0 / (SELECT COUNT(*) FROM Users), 2) AS percentage
FROM
    Register r
GROUP BY
    r.contest_id
ORDER BY
    percentage DESC,
    r.contest_id ASC;
```

## CROSS JOIN + GROUP BY
```
SELECT
    r.contest_id,
    ROUND(COUNT(*) * 100.0 / u.total_users, 2) AS percentage
FROM Register r
CROSS JOIN (
    SELECT COUNT(*) AS total_users
    FROM Users
) u
GROUP BY r.contest_id
ORDER BY percentage DESC, contest_id;
```

## CTE + GROUP BY
```
WITH total_users AS (
    SELECT COUNT(*) AS cnt
    FROM Users
)
SELECT
    r.contest_id,
    ROUND(COUNT(*) * 100.0 / t.cnt, 2) AS percentage
FROM Register r
CROSS JOIN total_users t
GROUP BY r.contest_id
ORDER BY percentage DESC, contest_id;
```

## Derived Table + GROUP BY
```
SELECT
    contest_id,
    ROUND(users_count * 100.0 / total_users, 2) AS percentage
FROM (
    SELECT
        contest_id,
        COUNT(*) AS users_count,
        (SELECT COUNT(*) FROM Users) AS total_users
    FROM Register
    GROUP BY contest_id
) t
ORDER BY percentage DESC, contest_id;
```
