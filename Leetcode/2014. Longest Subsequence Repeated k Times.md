# 2014. Longest Subsequence Repeated k Times

## DFS + Backtracking + Frequency Pruning - TLE
```
class Solution {
    String ans = "";
    public String longestSubsequenceRepeatedK(String s, int k) {
        int[] cnt = new int[26];
        for(char c : s.toCharArray())
            cnt[c - 'a']++;
        List<Character> chars = new ArrayList<>();
        for(int i = 25; i >= 0; i--) {
            for(int j = 0; j < cnt[i] / k; j++)
                chars.add((char)('a' + i));
        }
        dfs("", chars, s, k);
        return ans;
    }

    private void dfs(String cur, List<Character> chars, String s, int k) {
        if(cur.length() > ans.length())
            ans = cur;
        for(char c : chars) {
            String next = cur + c;
            if(isValid(next, s, k))
                dfs(next, chars, s, k);
        }
    }

    private boolean isValid(String seq, String s, int k) {
        int idx = 0;
        int repeat = 0;
        for(char ch : s.toCharArray()) {
            if(ch == seq.charAt(idx)) {
                idx++;
                if(idx == seq.length()) {
                    idx = 0;
                    repeat++;

                    if(repeat == k)
                        return true;
                }
            }
        }
        return false;
    }
}

## BFS + Frequency Pruning + Greedy Subsequence Check
```
class Solution {
    public String longestSubsequenceRepeatedK(String s, int k) {
        int[] cnt = new int[26];
        for(char c : s.toCharArray()) cnt[c - 'a']++;
        List<Character> chars = new ArrayList<>();
        for(char c = 'a'; c <= 'z'; c++) {
            if(cnt[c - 'a'] >= k) chars.add(c);
        }
        Queue<String> q = new LinkedList<>();
        q.offer("");
        String ans = "";
        while(!q.isEmpty()) {
            String cur = q.poll();
            for(char c : chars) {
                String nxt = cur + c;
                if(check(nxt, s, k)) {
                    ans = nxt;
                    q.offer(nxt);
                }
            }
        }
        return ans;
    }
    private boolean check(String t, String s, int k) {
        int i = 0;
        for(char c : s.toCharArray()) {
            if(c == t.charAt(i)) {
                i++;
                if(i == t.length()) {
                    k--;
                    if(k == 0) return true;
                    i = 0;
                }
            }
        }
        return false;
    }
}
