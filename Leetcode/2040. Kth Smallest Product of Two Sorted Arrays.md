# 2040. Kth Smallest Product of Two Sorted Arrays

## Binary Search on the Answer + Counting via Binary Search
```
class Solution {
    private static final long INF = 10000000000L;

    public long kthSmallestProduct(int[] nums1, int[] nums2, long k) {
        long low = -INF;
        long high = INF;

        while (low < high) {
            long mid = low + (high - low) / 2;

            if (count(nums1, nums2, mid) >= k) {
                high = mid;
            } else {
                low = mid + 1;
            }
        }

        return low;
    }

    private long count(int[] nums1, int[] nums2, long target) {
        long count = 0;
        int n = nums2.length;

        for (int a : nums1) {
            if (a > 0) {
                int left = 0, right = n - 1;
                int ans = -1;

                while (left <= right) {
                    int mid = left + (right - left) / 2;

                    if ((long) a * nums2[mid] <= target) {
                        ans = mid;
                        left = mid + 1;
                    } else {
                        right = mid - 1;
                    }
                }

                count += ans + 1;
            } else if (a < 0) {
                int left = 0, right = n - 1;
                int ans = n;

                while (left <= right) {
                    int mid = left + (right - left) / 2;

                    if ((long) a * nums2[mid] <= target) {
                        ans = mid;
                        right = mid - 1;
                    } else {
                        left = mid + 1;
                    }
                }

                count += n - ans;
            } else {
                if (target >= 0) {
                    count += n;
                }
            }
        }

        return count;
    }
}
```

## Binary Search on the Answer (Parametric Search) + Binary Search Counting
```
class Solution {
    static long inf = (long) 1e10;

    public long kthSmallestProduct(int[] nums1, int[] nums2, long k) {
        int n = nums2.length;
        long lo = -inf - 1;
        long hi = inf + 1;
        while (lo < hi) {
            long mid = lo + ((hi - lo) >> 1);
            long cnt = 0;
            for (int i : nums1) {
                int l = 0;
                int r = n - 1;
                int p = 0;
                if (0 <= i) {
                    while (l <= r) {
                        int c = l + ((r - l) >> 1);
                        long mul = i * (long) nums2[c];
                        if (mul <= mid) {
                            p = c + 1;
                            l = c + 1;
                        } else {
                            r = c - 1;
                        }
                    }
                } else {
                    while (l <= r) {
                        int c = l + ((r - l) >> 1);
                        long mul = i * (long) nums2[c];
                        if (mul <= mid) {
                            p = n - c;
                            r = c - 1;
                        } else {
                            l = c + 1;
                        }
                    }
                }
                cnt += p;
            }
            if (cnt >= k) {
                hi = mid;
            } else {
                lo = mid + 1L;
            }
        }
        return lo;
    }
}
```
