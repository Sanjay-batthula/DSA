# 2140. Solving Questions With Brainpower

## Brute Force - TLE
```
class Solution {
    private int[][] questions;
    private int n;
    public long mostPoints(int[][] questions) {
        this.questions = questions;
        n = questions.length;
        return dfs(0);
    }
    private long dfs(int i) {
        if (i >= n) {
            return 0;
        }
        long solve = questions[i][0] + dfs(i + questions[i][1] + 1);
        long skip = dfs(i + 1);
        return Math.max(solve, skip);
    }
}
```


## Dynamic Programming with Memoization (Top-down)
```
class Solution {
    private int n;
    private Long[] f;
    private int[][] questions;

    public long mostPoints(int[][] questions) {
        n = questions.length;
        f = new Long[n];
        this.questions = questions;
        return dfs(0);
    }

    private long dfs(int i) {
        if (i >= n) {
            return 0;
        }
        if (f[i] != null) {
            return f[i];
        }
        int p = questions[i][0], b = questions[i][1];
        return f[i] = Math.max(p + dfs(i + b + 1), dfs(i + 1));
    }
}
```

## Bottom-Up Dynamic Programming (Tabulation)
```
class Solution {
    public long mostPoints(int[][] questions) {
        int n = questions.length;
        long[] dp = new long[n + 1];
        for (int i = n - 1; i >= 0; i--) {
            long solve = questions[i][0];
            int next = i + questions[i][1] + 1;
            if (next < n) {
                solve += dp[next];
            }
            long skip = dp[i + 1];
            dp[i] = Math.max(solve, skip);
        }
        return dp[0];
    }
}
```
