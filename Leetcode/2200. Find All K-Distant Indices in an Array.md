# 2200. Find All K-Distant Indices in an Array

## Brute Force
```
class Solution {
    public List<Integer> findKDistantIndices(int[] nums, int key, int k) {
        int n = nums.length;
        List<Integer> ans = new ArrayList<>();
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                if (Math.abs(i - j) <= k && nums[j] == key) {
                    ans.add(i);
                    break;
                }
            }
        }
        return ans;
    }
}
```

## Range Expansion (Greedy Interval Expansion)
```
class Solution {
    public List<Integer> findKDistantIndices(int[] nums, int key, int k) {
        int n = nums.length;
        List<Integer> ans = new ArrayList<>();
        int start = 0;
        for(int i=0;i<n;i++){
            if(nums[i]==key){
                int left=Math.max(start,i-k);
                int right=Math.min(n-1,i+k);
                for(int j=left;j<=right;j++){
                    ans.add(j);
                }
                start=right+1;
            }
        }
        return ans;
    }
}
```

## Two Pointers / Sliding Window
```
class Solution {
    public List<Integer> findKDistantIndices(int[] nums, int key, int k) {
        int n=nums.length;
        List<Integer> ans=new ArrayList<>();
        List<Integer> keys=new ArrayList<>();
        for(int i=0;i<n;i++){
            if(nums[i]==key) keys.add(i);
        }
        int p=0;
        for(int i=0;i<n;i++){
            while(p<keys.size() && keys.get(p)<i-k) p++;
            if(p<keys.size() && Math.abs(keys.get(p)-i)<=k) ans.add(i);
        }
        return ans;
    }
}
```
