# 2215. Find the Difference of Two Arrays

## HashSet / Set Difference Approach
```
import java.util.*;

class Solution {
    public List<List<Integer>> findDifference(int[] nums1, int[] nums2) {
        Set<Integer> set1 = new HashSet<>();
        Set<Integer> set2 = new HashSet<>();
        for (int num : nums1) {
            set1.add(num);
        }
        for (int num : nums2) {
            set2.add(num);
        }
        Set<Integer> uniqueInNums1 = new HashSet<>(set1);
        uniqueInNums1.removeAll(set2); // Elements in nums1 but not in nums2
        
        Set<Integer> uniqueInNums2 = new HashSet<>(set2);
        uniqueInNums2.removeAll(set1); // Elements in nums2 but not in nums1
        
        List<List<Integer>> result = new ArrayList<>();
        result.add(new ArrayList<>(uniqueInNums1));
        result.add(new ArrayList<>(uniqueInNums2));
        
        return result;
    }
}

## Boolean Array / Frequency Map (Bucket) Approach
```
import java.util.*;

class Solution {
    public List<List<Integer>> findDifference(int[] nums1, int[] nums2) {
        boolean[] seen1 = new boolean[2001];
        boolean[] seen2 = new boolean[2001];
        
        for (int n : nums1) seen1[n + 1000] = true;
        for (int n : nums2) seen2[n + 1000] = true;
        
        List<Integer> list1 = new ArrayList<>();
        List<Integer> list2 = new ArrayList<>();
        for (int i = 0; i < 2001; i++) {
            if (seen1[i] && !seen2[i]) list1.add(i - 1000);
            if (seen2[i] && !seen1[i]) list2.add(i - 1000);
        }
        List<List<Integer>> result = new ArrayList<>();
        result.add(list1);
        result.add(list2);
        return result;
    }
}
