# 2375. Construct Smallest Number From DI String

## Backtracking (DFS with Pruning)
```
class Solution {
    String ans = "";
    boolean[] vis = new boolean[10];
    public String smallestNumber(String pattern) {
        dfs(pattern, new StringBuilder());
        return ans;
    }

    private void dfs(String pattern, StringBuilder temp) {
        if (!ans.isEmpty()) {
            return;
        }

        if (temp.length() == pattern.length() + 1) {
            ans = temp.toString();
            return;
        }

        for (int i = 1; i <= 9; i++) {
            if (!vis[i]) {
                if (temp.length() > 0) {
                    char prev = temp.charAt(temp.length() - 1);

                    if (pattern.charAt(temp.length() - 1) == 'I' && prev >= (char)(i + '0')) {
                        continue;
                    }

                    if (pattern.charAt(temp.length() - 1) == 'D' && prev <= (char)(i + '0')) {
                        continue;
                    }
                }

                vis[i] = true;
                temp.append(i);

                dfs(pattern, temp);

                temp.deleteCharAt(temp.length() - 1);
                vis[i] = false;
            }
        }
    }
}
```

## Monotonic Stack + Greedy Construction
```
class Solution {
    public String smallestNumber(String pattern) {
        StringBuilder ans = new StringBuilder();
        Stack<Integer> stack = new Stack<>();
        for (int i = 0; i <= pattern.length(); i++) {
            stack.push(i + 1);
            if (i == pattern.length() || pattern.charAt(i) == 'I') {
                while (!stack.isEmpty()) {
                    ans.append(stack.pop());
                }
            }
        }
        return ans.toString();
    }
}
```
