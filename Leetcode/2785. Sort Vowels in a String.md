## Brute Force

```
class Solution {
    public String sortVowels(String s) {
        String vowels = "aeiouAEIOU";
        List<Character> list = new ArrayList<>();
        
        for (char c : s.toCharArray()) {
            if (vowels.indexOf(c) != -1) list.add(c);
        }
        
        Collections.sort(list);
        
        StringBuilder sb = new StringBuilder();
        int idx = 0;
        for (char c : s.toCharArray()) {
            if (vowels.indexOf(c) != -1) {
                sb.append(list.get(idx++));
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }
}
```

## Counting Sort
```
class Solution {
    public String sortVowels(String s) {
        String vowels = "aeiouAEIOU";
        int[] freq = new int[128]; 
        for (char c : s.toCharArray()) {
            if (vowels.indexOf(c) != -1) freq[c]++;
        }
        StringBuilder sb = new StringBuilder();
        for (char c : s.toCharArray()) {
            if (vowels.indexOf(c) != -1) {
                for (int ascii = 0; ascii < 128; ascii++) {
                    if (freq[ascii] > 0) {
                        sb.append((char) ascii);
                        freq[ascii]--;
                        break;
                    }
                }
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }
}
```


## 10 Bucket version - Bucketed Counting Sort with Fixed Alphabet

```
class Solution {
    public String sortVowels(String s) {
        char[] order = {'A','E','I','O','U','a','e','i','o','u'};
        int[] freq = new int[order.length];
        String vowels = "AEIOUaeiou";
        
        for (char c : s.toCharArray()) {
            int idx = vowels.indexOf(c);
            if (idx != -1) freq[idx]++;
        }
        
        StringBuilder sb = new StringBuilder();
        int p = 0; 
        for (char c : s.toCharArray()) {
            if (vowels.indexOf(c) != -1) {
                while (p < order.length && freq[p] == 0) p++;
                sb.append(order[p]);
                freq[p]--;
            } else {
                sb.append(c);
            }
        }
        return sb.toString();
    }
}
```

        for (char c : s.toCharArray()) {

                sb.append(c);
            }
        }
