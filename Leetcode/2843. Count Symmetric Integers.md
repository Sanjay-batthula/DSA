# 2843. Count Symmetric Integers

## Enumeration with digit processing
```
class Solution {
    public int countSymmetricIntegers(int low, int high) {
        int ans = 0;
        for (int x = low; x <= high; ++x) {
            ans += f(x);
        }
        return ans;
    }
    private int f(int x) {
        String s = "" + x;
        int n = s.length();
        if (n % 2 == 1) {
            return 0;
        }
        int a = 0, b = 0;
        for (int i = 0; i < n / 2; ++i) {
            a += s.charAt(i) - '0';
        }
        for (int i = n / 2; i < n; ++i) {
            b += s.charAt(i) - '0';
        }
        return a == b ? 1 : 0;
    }
}
```

## Arithmetic digit extraction
```
class Solution {
    public int countSymmetricIntegers(int low, int high) {
        int ans = 0;
        for (int x = low; x <= high; x++) {
            int d = digits(x);
            if (d % 2 == 1) continue;
            int half = d / 2;
            int left = 0, right = 0;
            int pow = (int)Math.pow(10, half);
            int l = x / pow;
            int r = x % pow;
            while (l > 0) {
                left += l % 10;
                l /= 10;
            }
            while (r > 0) {
                right += r % 10;
                r /= 10;
            }
            if (left == right) ans++;
        }
        return ans;
    }

    private int digits(int x) {
        int c = 0;
        while (x > 0) {
            c++;
            x /= 10;
        }
        return c;
    }
}
```
