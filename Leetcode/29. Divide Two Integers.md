## Bit Manipulation + Exponential Search approach
```
class Solution {
    public int divide(int dividend, int divisor) {
        if (dividend == Integer.MIN_VALUE && divisor == -1) return Integer.MAX_VALUE;
        if (divisor == 1) return dividend;

        boolean sameSign = (dividend >= 0 && divisor >= 0) || (dividend <= 0 && divisor <= 0);
        long a = Math.abs((long) dividend);
        long b = Math.abs((long) divisor);
        int ans = 0;

        while (a >= b) {
            long temp = b, mul = 1;
            while ((temp << 1) <= a) {
                temp <<= 1;
                mul <<= 1;
            }
            a -= temp;
            ans += mul;
        }

        return sameSign ? ans : -ans;
    }
}
```


## Subtraction Approach (Brute Force) -TLE
```
class Solution {
    public int divide(int dividend, int divisor) {
        if (dividend == Integer.MIN_VALUE && divisor == -1) return Integer.MAX_VALUE;
        boolean sign = (dividend < 0) ^ (divisor < 0);
        long a = Math.abs((long) dividend);
        long b = Math.abs((long) divisor);
        int res = 0;
        while (a >= b) {
            a -= b;
            res++;
        }
        return sign ? -res : res;
    }
}
```

## Binary Search on Quotient -TLE
```
class Solution {
    long multiply(long a, long b) {
        long res = 0;
        while (b > 0) {
            res += a;
            b--;
        }
        return res;
    }

    public int divide(int dividend, int divisor) {
        if (dividend == Integer.MIN_VALUE && divisor == -1) return Integer.MAX_VALUE;
        boolean sign = (dividend < 0) ^ (divisor < 0);
        long a = Math.abs((long) dividend);
        long b = Math.abs((long) divisor);
        long left = 0, right = a, ans = 0;

        while (left <= right) {
            long mid = left + (right - left) / 2;
            long val = multiply(b, mid);
            if (val <= a) {
                ans = mid;
                left = mid + 1;
            } else right = mid - 1;
        }

        if (sign) ans = -ans;
        return (int) ans;
    }
}
```

## Bit-by-Bit Division (Simulated Long Division)
```
class Solution {
    public int divide(int dividend, int divisor) {
        if (dividend == Integer.MIN_VALUE && divisor == -1) return Integer.MAX_VALUE;

        long a = Math.abs((long) dividend);
        long b = Math.abs((long) divisor);
        int res = 0;

        for (int i = 31; i >= 0; i--) {
            if ((a >> i) >= b) {
                a -= b << i;
                res += 1 << i;
            }
        }
        if ((dividend < 0) ^ (divisor < 0)) res = -res;
        return res;
    }
}
```

## Recursive Division via Doubling
```
class Solution {
    long solve(long a, long b) {
        if (a < b) return 0;
        long sum = b, res = 1;
        while ((sum + sum) <= a) {
            sum += sum;
            res += res;
        }
        return res + solve(a - sum, b);
    }

    public int divide(int dividend, int divisor) {
        if (dividend == Integer.MIN_VALUE && divisor == -1) return Integer.MAX_VALUE;
        boolean sign = (dividend < 0) ^ (divisor < 0);
        long a = Math.abs((long) dividend);
        long b = Math.abs((long) divisor);
        long ans = solve(a, b);
        return sign ? (int)-ans : (int)ans;
    }
}
```
