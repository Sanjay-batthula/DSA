# 3356. Zero Array Transformation II

## Binary Search on Answer + Difference Array
```java
class Solution {
    private int n;
    private int[] nums;
    private int[][] queries;

    public int minZeroArray(int[] nums, int[][] queries) {
        this.nums = nums;
        this.queries = queries;
        n = nums.length;
        int m = queries.length;
        int l = 0, r = m + 1;
        while (l < r) {
            int mid = (l + r) >> 1;
            if (check(mid)) {
                r = mid;
            } else {
                l = mid + 1;
            }
        }
        return l > m ? -1 : l;
    }

    private boolean check(int k) {
        int[] d = new int[n + 1];
        for (int i = 0; i < k; ++i) {
            int l = queries[i][0], r = queries[i][1], val = queries[i][2];
            d[l] += val;
            d[r + 1] -= val;
        }
        for (int i = 0, s = 0; i < n; ++i) {
            s += d[i];
            if (nums[i] > s) {
                return false;
            }
        }
        return true;
    }
}
```

## Greedy Query Processing + Difference Array
```java
class Solution {
    public int minZeroArray(int[] nums, int[][] queries) {
        int n = nums.length;
        int[] diff = new int[n + 1];
        int sum = 0;
        int q = 0;
        for (int i = 0; i < n; i++) {
            sum += diff[i];
            while (sum < nums[i]) {
                if (q == queries.length) {
                    return -1;
                }
                int l = queries[q][0];
                int r = queries[q][1];
                int val = queries[q][2];
                q++;
                if (r < i) {
                    continue;
                }
                if (l <= i) {
                    sum += val;
                    diff[r + 1] -= val;
                } else {
                    diff[l] += val;
                    diff[r + 1] -= val;
                }
            }
        }
        return q;
    }
}
```
