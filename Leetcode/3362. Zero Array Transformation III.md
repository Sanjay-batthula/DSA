# 3362. Zero Array Transformation III

## Brute Force / Backtracking
```py
class Solution:
    def maxRemoval(self, nums: List[int], queries: List[List[int]]) -> int:
        n = len(queries)
        def can_make_zero(mask):
            coverage = [0] * len(nums)
            for j in range(n):
                if mask & (1 << j):
                    l, r = queries[j]
                    for i in range(l, r + 1):
                        coverage[i] += 1
            for i in range(len(nums)):
                if coverage[i] < nums[i]:
                    return False
            return True
        best = 0
        for mask in range(1 << n):
            if can_make_zero(mask):
                best = max(best, n - mask.bit_count())
        return best
```

## greedy + max-heap + difference array
```py
class Solution:
    def maxRemoval(self, nums: List[int], queries: List[List[int]]) -> int:
        queries.sort()
        pq = []
        d = [0] * (len(nums) + 1)
        s = j = 0
        for i, x in enumerate(nums):
            s += d[i]
            while j < len(queries) and queries[j][0] <= i:
                heappush(pq, -queries[j][1])
                j += 1
            while s < x and pq and -pq[0] >= i:
                s += 1
                d[-heappop(pq) + 1] -= 1
            if s < x:
                return -1
        return len(pq)
```
