# 3394. Check if Grid can be Cut into Sections

## Sweep Line / Event Line Algorithm (Python)
```
from typing import List

class Solution:
    def countLineIntersections(self, coordinates: List[tuple]) -> bool:
        lines = 0
        overlap = 0
        for coord in coordinates:
            if coord[1] == 0:
                overlap -= 1
            else:
                overlap += 1
            if overlap == 0:
                lines += 1
        return lines >= 3

    def checkValidCuts(self, n: int, rectangles: List[List[int]]) -> bool:
        y_coordinates = []
        x_coordinates = []

        for rectangle in rectangles:
            y_coordinates.append((rectangle[1], 1))
            y_coordinates.append((rectangle[3], 0))
            x_coordinates.append((rectangle[0], 1))
            x_coordinates.append((rectangle[2], 0))
        y_coordinates.sort()
        x_coordinates.sort()
        return self.countLineIntersections(y_coordinates) or self.countLineIntersections(x_coordinates)
```

## Interval Merging (Java)
```
import java.util.*;

class Solution {
    public boolean checkValidCuts(int n, int[][] rectangles) {
        return countSections(rectangles, 0, 2) >= 3 || countSections(rectangles, 1, 3) >= 3;
    }

    private int countSections(int[][] rectangles, int startIdx, int endIdx) {
        int m = rectangles.length;
        int[][] intervals = new int[m][2];

        for (int i = 0; i < m; i++) {
            intervals[i][0] = rectangles[i][startIdx];
            intervals[i][1] = rectangles[i][endIdx];
        }

        Arrays.sort(intervals, (a, b) -> {
            if (a[0] != b[0]) return Integer.compare(a[0], b[0]);
            return Integer.compare(a[1], b[1]);
        });

        int sections = 0;
        int end = -1;

        for (int[] in : intervals) {
            if (sections == 0) {
                sections = 1;
                end = in[1];
            } else if (in[0] < end) {
                end = Math.max(end, in[1]);
            } else {
                sections++;
                end = in[1];
            }
        }

        return sections;
    }
}
```
