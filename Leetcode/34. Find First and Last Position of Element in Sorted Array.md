## Brute Force
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        int[] result = {-1, -1};
        for (int i = 0; i < nums.length; i++) {
            if (nums[i] == target) {
                result[0] = i;
                break;
            }
        }
        if (result[0] == -1) {
            return result;
        }
        for (int i = nums.length - 1; i >= 0; i--) {
            if (nums[i] == target) {
                result[1] = i;
                break;
            }
        }
        return result;
    }
}
```

# Binary Search
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        int leftIndex = findFirst(nums, target);
        int rightIndex = findFirst(nums, target + 1);
        if (leftIndex == rightIndex) {
            return new int[] {-1, -1};
        } else {
            return new int[] {leftIndex, rightIndex - 1};
        }
    }

    private int findFirst(int[] nums, int x) {
        int left = 0, right = nums.length;
        while (left < right) {
            int mid = (left + right) >>> 1;
            if (nums[mid] >= x) {
                right = mid;
            } else {
                left = mid + 1;
            }
        }
        return left;
    }
}
```

## HashMap Frequency + First Occurrence Scan
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        HashMap<Integer, ArrayList<Integer>> map = new HashMap<>();
        for (int i = 0; i < nums.length; i++) {
            map.computeIfAbsent(nums[i], k -> new ArrayList<>()).add(i);
        }
        if (!map.containsKey(target)) return new int[]{-1, -1};
        ArrayList<Integer> arr = map.get(target);
        return new int[]{arr.get(0), arr.get(arr.size() - 1)};
    }
}
```

## Binary Search for Left + Binary Search for Right (Classic Double-Search)
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        return new int[]{first(nums, target), last(nums, target)};
    }

    int first(int[] a, int t) {
        int l = 0, r = a.length - 1, ans = -1;
        while (l <= r) {
            int m = (l + r) >>> 1;
            if (a[m] >= t) r = m - 1;
            else l = m + 1;
            if (a[m] == t) ans = m;
        }
        return ans;
    }

    int last(int[] a, int t) {
        int l = 0, r = a.length - 1, ans = -1;
        while (l <= r) {
            int m = (l + r) >>> 1;
            if (a[m] <= t) l = m + 1;
            else r = m - 1;
            if (a[m] == t) ans = m;
        }
        return ans;
    }
}
```

## Built-in Arrays.binarySearch Trick - Wrong Answer
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        int left = lowerBound(nums, target);
        if (left == nums.length || nums[left] != target) return new int[]{-1, -1};

        int right = upperBound(nums, target) - 1;
        return new int[]{left, right};
    }

    int lowerBound(int[] a, int t) {
        int i = Arrays.binarySearch(a, t);
        return i < 0 ? -i - 1 : i;
    }

    int upperBound(int[] a, int t) {
        int i = Arrays.binarySearch(a, t);
        if (i < 0) return -i - 1;
        while (i < a.length && a[i] == t) i++;
        return i;
    }
}
```

## Expansion From Any Found Index (Optimized Hybrid)
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        int idx = Arrays.binarySearch(nums, target);
        if (idx < 0) return new int[]{-1, -1};

        int l = idx, r = idx;
        while (l > 0 && nums[l - 1] == target) l--;
        while (r < nums.length - 1 && nums[r + 1] == target) r++;

        return new int[]{l, r};
    }
}
```

## Prefix Boundary Binary Search
```
class Solution {
    public int[] searchRange(int[] nums, int target) {
        int left = boundary(nums, target);
        int right = boundary(nums, target + 1) - 1;

        if (left == nums.length || nums[left] != target) return new int[]{-1, -1};
        return new int[]{left, right};
    }

    int boundary(int[] a, int t) {
        int l = 0, r = a.length;
        while (l < r) {
            int m = (l + r) >>> 1;
            if (a[m] >= t) r = m;
            else l = m + 1;
        }
        return l;
    }
}
```
