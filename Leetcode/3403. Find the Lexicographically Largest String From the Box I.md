## Brute Force
```
class Solution {
    public String answerString(String word, int numFriends) {
        if (numFriends == 1) {
            return word;
        }
        int n = word.length();
        int maxLen = n - numFriends + 1;
        String ans = "";
        for (int i = 0; i < n; i++) {
            int len = Math.min(maxLen, n - i);
            String cur = word.substring(i, i + len);
            if (cur.compareTo(ans) > 0) {
                ans = cur;
            }
        }
        return ans;
    }
}
```

## Two-Pointer Maximum Suffix Algorithm
```
class Solution {
    public String answerString(String word, int numFriends) {
        if (numFriends == 1) {
            return word;
        }
        String s = lastSubstring(word);
        return s.substring(0, Math.min(s.length(), word.length() - numFriends + 1));
    }

    public String lastSubstring(String s) {
        int n = s.length();
        int i = 0, j = 1, k = 0;
        while (j + k < n) {
            int d = s.charAt(i + k) - s.charAt(j + k);
            if (d == 0) {
                ++k;
            } else if (d < 0) {
                i += k + 1;
                k = 0;
                if (i >= j) {
                    j = i + 1;
                }
            } else {
                j += k + 1;
                k = 0;
            }
        }
        return s.substring(i);
    }
}
```

## Suffix Array
```
import java.util.*;

class Solution {
    public String answerString(String word, int numFriends) {
        if (numFriends == 1) {
            return word;
        }

        int n = word.length();
        int[] sa = buildSuffixArray(word);

        int start = sa[n - 1];
        int len = Math.min(n - start, n - numFriends + 1);

        return word.substring(start, start + len);
    }

    private int[] buildSuffixArray(String s) {
        int n = s.length();

        Integer[] sa = new Integer[n];
        int[] rank = new int[n];
        int[] temp = new int[n];

        for (int i = 0; i < n; i++) {
            sa[i] = i;
            rank[i] = s.charAt(i);
        }
        for (int k = 1; k < n; k <<= 1) {
            final int gap = k;
            final int[] r = rank;
            Arrays.sort(sa, (a, b) -> {
                if (r[a] != r[b]) {
                    return r[a] - r[b];
                }
                int ra = (a + gap < n) ? r[a + gap] : -1;
                int rb = (b + gap < n) ? r[b + gap] : -1;
                return ra - rb;
            });
            temp[sa[0]] = 0;
            for (int i = 1; i < n; i++) {
                int prev = sa[i - 1];
                int curr = sa[i];
                boolean same = rank[prev] == rank[curr];
                if (same) {
                    int r1 = (prev + gap < n) ? rank[prev + gap] : -1;
                    int r2 = (curr + gap < n) ? rank[curr + gap] : -1;
                    same = r1 == r2;
                }
                temp[curr] = temp[prev] + (same ? 0 : 1);
            }
            System.arraycopy(temp, 0, rank, 0, n);
            if (rank[sa[n - 1]] == n - 1) {
                break;
            }
        }
        int[] res = new int[n];
        for (int i = 0; i < n; i++) {
            res[i] = sa[i];
        }
        return res;
    }
}
```
