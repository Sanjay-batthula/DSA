# 3411. Maximum Subarray With Equal Products

##
```
class Solution {
    public int maxLength(int[] nums) {
        int n = nums.length;
        int maxLength = 0;

        for (int i = 0; i < n; i++) {
            long prod = 1;
            int gcd = 0;
            long lcm = 1;

            for (int j = i; j < n; j++) {
                prod *= nums[j];
                gcd = gcd == 0 ? nums[j] : gcd(gcd, nums[j]);
                lcm = (lcm * nums[j]) / gcd(gcd((int) lcm, nums[j]), nums[j]);

                if (prod == gcd * lcm) {
                    maxLength = Math.max(maxLength, j - i + 1);
                }
            }
        }

        return maxLength;
    }

    private int gcd(int a, int b) {
        while (b != 0) {
            int temp = b;
            b = a % b;
            a = temp;
        }
        return a;
    }
}
```

```
class Solution {
    public int maxLength(int[] nums) {
        int n = nums.length;
        int ans = 2;

        for (int i = 0; i < n; i++) {
            int g = 0;

            for (int j = i; j < n; j++) {
                g = (g == 0) ? nums[j] : gcd(g, nums[j]);

                boolean ok = true;

                for (int x = i; x <= j && ok; x++) {
                    for (int y = x + 1; y <= j; y++) {
                        if (gcd(nums[x] / g, nums[y] / g) != 1) {
                            ok = false;
                            break;
                        }
                    }
                }

                if (ok)
                    ans = Math.max(ans, j - i + 1);
            }
        }

        return ans;
    }

    private int gcd(int a, int b) {
        while (b != 0) {
            int t = a % b;
            a = b;
            b = t;
        }
        return a;
    }
}
```
