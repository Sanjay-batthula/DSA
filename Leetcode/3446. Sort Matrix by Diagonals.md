# 3446. Sort Matrix by Diagonals

## Diagonal Traversal + Sorting (Simulation)
```java
class Solution {
    public int[][] sortMatrix(int[][] grid) {
        int n = grid.length;

        for (int i = 0; i < n; i++) {
            List<Integer> diag = new ArrayList<>();
            int r = i, c = 0;
            while (r < n && c < n) {
                diag.add(grid[r][c]);
                r++;
                c++;
            }
            diag.sort(Collections.reverseOrder());
            r = i; c = 0;
            int idx = 0;
            while (r < n && c < n) {
                grid[r][c] = diag.get(idx++);
                r++;
                c++;
            }
        }

        for (int j = 1; j < n; j++) {
            List<Integer> diag = new ArrayList<>();
            int r = 0, c = j;
            while (r < n && c < n) {
                diag.add(grid[r][c]);
                r++;
                c++;
            }
            Collections.sort(diag);
            r = 0; c = j;
            int idx = 0;
            while (r < n && c < n) {
                grid[r][c] = diag.get(idx++);
                r++;
                c++;
            }
        }

        return grid;
    }
}
```

class Solution:
    def sortMatrix(self, grid: List[List[int]]) -> List[List[int]]:
        n = len(grid)
        for r in range(n):
            diag = []
            i, j = r, 0
            while i < n and j < n:
                diag.append(grid[i][j])
                i += 1
                j += 1
            diag.sort(reverse=True)
            i, j = r, 0
            k = 0
            while i < n and j < n:
                grid[i][j] = diag[k]
                k += 1
                i += 1
                j += 1
        for c in range(1, n):
            diag = []
            i, j = 0, c
            while i < n and j < n:
                diag.append(grid[i][j])
                i += 1
                j += 1
            diag.sort()
            i, j = 0, c
            k = 0
            while i < n and j < n:
                grid[i][j] = diag[k]
                k += 1
                i += 1
                j += 1
        return grid
