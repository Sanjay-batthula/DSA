# 3447. Assign Elements to Groups with Constraints


## Brute Force - TLE
```
class Solution:
    def assignElements(self, groups: List[int], elements: List[int]) -> List[int]:
        ans = []
        for group in groups:
            idx = -1
            for i, element in enumerate(elements):
                if group % element == 0:
                    idx = i
                    break
            ans.append(idx)
        return ans
```

## Hash Map + Divisor Enumeration
```
class Solution {
    public int[] assignElements(int[] groups, int[] elements) {
        int n = groups.length;
        int m = elements.length;
        int[] assigned= new int[n];
        Map<Integer ,Integer> elementMap= new HashMap<>();
            for(int j=0;j<m;j++){
            if(!elementMap.containsKey(elements[j])){
                elementMap.put(elements[j],j);
            }
            }
        for(int i=0;i<n;i++){
            int groupSize=groups[i];
            int minIndex=Integer.MAX_VALUE;
            for(int d=1;d*d<=groupSize;d++){
                if(groupSize %d == 0){
                    if(elementMap.containsKey(d)){
                        minIndex=Math.min(minIndex,elementMap.get(d));
                    }
                    if(d!=groupSize/d && elementMap.containsKey(groupSize/d)){
                        minIndex=Math.min(minIndex,elementMap.get(groupSize/d));
                    }
                }
            }
            assigned[i]=(minIndex!=Integer.MAX_VALUE)?minIndex:-1;
        }
        return assigned;
        
    }
}
```

## Hash Map + Sieve (Multiples Enumeration) - TLE
```
class Solution:
    def assignElements(self, groups: List[int], elements: List[int]) -> List[int]:
        MAX = max(groups)
        best = [-1] * (MAX + 1)
        for i, e in enumerate(elements):
            for multiple in range(e, MAX + 1, e):
                if best[multiple] == -1:
                    best[multiple] = i
        return [best[g] for g in groups]
```
