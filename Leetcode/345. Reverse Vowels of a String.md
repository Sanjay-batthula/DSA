## Brute Force
```
class Solution {
    public String reverseVowels(String s) {
        ArrayList<Character> vowels = new ArrayList<>();
        char[] arr = s.toCharArray();
        for (char c : arr) {
            if (isVowel(c)) vowels.add(c);
        }
        int idx = vowels.size() - 1;
        for (int i = 0; i < arr.length; i++) {
            if (isVowel(arr[i])) {
                arr[i] = vowels.get(idx--);
            }
        }
        return new String(arr);
    }
    private boolean isVowel(char c) {
        return "aeiouAEIOU".indexOf(c) != -1;
    }
}
```

## HashSet
```
class Solution {
    private static final Set<Character> vowels =
        new HashSet<>(Arrays.asList(
            'a','e','i','o','u','A','E','I','O','U'
        ));
    public String reverseVowels(String s) {
        char[] arr = s.toCharArray();
        int l = 0, r = arr.length - 1;
        while (l < r) {
            if (!vowels.contains(arr[l])) l++;
            else if (!vowels.contains(arr[r])) r--;
            else {
                char t = arr[l];
                arr[l] = arr[r];
                arr[r] = t;
                l++; r--;
            }
        }
        return new String(arr);
    }
}
```

## Two Pointers approach
```
class Solution {
    public String reverseVowels(String s) {
        int left = 0;
        int right = s.length() - 1;
        char[] arr = s.toCharArray();

        while (left < right) {
            if (!isVowel(arr[left])) {
                left++;
            } else if (!isVowel(arr[right])) {
                right--;
            } else {
                char temp = arr[left];
                arr[left] = arr[right];
                arr[right] = temp;
                left++;
                right--;
            }
        }

        return new String(arr);
    }

    private boolean isVowel(char c) {
        return "aeiouAEIOU".indexOf(c) != -1;
    }

    public static void main(String[] args) {
        Solution solution = new Solution();
        String s1 = "IceCreAm";
        String s2 = "leetcode";

        System.out.println(solution.reverseVowels(s1)); // Output: "AceCreIm"
        System.out.println(solution.reverseVowels(s2)); // Output: "leotcede"
    }
}
```
