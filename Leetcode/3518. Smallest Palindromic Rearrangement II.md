# 3518. Smallest Palindromic Rearrangement II

## Count characters and build half-frequency
```py
class Solution:
    def smallestPalindrome(self, s: str, k: int) -> str:
        from math import comb

        n = len(s)
        cnt = [0] * 26

        for c in s:
            cnt[ord(c) - ord('a')] += 1

        half = [x // 2 for x in cnt]
        m = sum(half)

        def ways(counts, total, limit):
            res = 1
            used = 0

            for x in counts:
                if x:
                    res *= comb(used + x, x)
                    used += x
                    if res >= limit:
                        return limit

            return res

        if ways(half, m, k) < k:
            return ""

        left = []

        for pos in range(m):
            for c in range(26):
                if half[c] == 0:
                    continue

                half[c] -= 1
                w = ways(half, m - pos - 1, k)

                if k > w:
                    k -= w
                    half[c] += 1
                else:
                    left.append(chr(ord('a') + c))
                    break

        left = ''.join(left)

        middle = ""
        for i in range(26):
            if cnt[i] % 2:
                middle = chr(ord('a') + i)
                break

        return left + middle + left[::-1]
```


## Count distinct permutations using combinatorics
```c++
#include<bits/stdc++.h>
using namespace std;
typedef unsigned long long ull;
const ull TH = 1e6+1;

ull cmb(int n, int r){
    ull res=1;
    for(int i=1;i<=r;i++){
        res=res*(n-r+i)/i;
        if(res>TH) return TH;
    }
        return res;
}

ull calc(vector<int>& a, int tot){
    ull res=1;
    for(int i=0;i<26;i++){
        if(a[i]){
            res*=cmb(tot,a[i]);
            if(res>TH) res=TH;
            tot-=a[i];        
        } 
    }
    return res;
}
class Solution {
public:
    string smallestPalindrome(string s, int k) {
        int f[26]{},h[26]{}, tot=0; char m = 0;
        for(char c: s) f[c-'a']++;
        for(int i=0;i<26;i++) h[i] = f[i]/2, tot +=h[i], m = f[i]%2 ? 'a'+i : m;
        vector<int> a(h,h+26);
        if(k>calc(a,tot)) return "";
        string r;
        while(tot--){
            for(int i=0;i<26;i++){
                if(!a[i]) continue;
                a[i]--;
                if(calc(a,tot) >=k) { r+='a'+i; break; }
                k-=calc(a,tot), a[i]++;
            }
        }
        string rr=r; reverse(rr.begin(), rr.end());
        return r + (m?string(1,m):"")+rr;
    }
};
```
