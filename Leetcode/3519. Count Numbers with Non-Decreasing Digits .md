# 3519. Count Numbers with Non-Decreasing Digits 

##
```c++
class Solution {
public:
    using ll = long long; const ll M=1e9+7;
    vector<int> cv(string s, int b){
        vector<int> v;
        while(s!="0"){
            int r=0; string ns;
            for(char c:s){
                int cur = r*10 +c-'0', q=cur/b;
                r=cur%b;
                if(!ns.empty() || q) ns+=q+'0';
            }
            v.push_back(r); s=ns.empty() ? "0" :ns;
        }
        reverse(v.begin(), v.end());
        return v.empty()? vector<int>{0} :v;
    }

    string ds(string s){
        for(int i= s.size()-1; i>=0;i--){
            if(s[i]!='0') { s[i]--; break; }
            s[i]='9';
        }
        auto pos = s.find_first_not_of('0');
        return pos == string::npos ? "0" : s.substr(pos);
    }

    ll dp(int i, int t, int st, int p, vector<int>& d, int b, auto& m){
        if(i==d.size()) return 1;
        if(m[i][t][st][p]!=-1) return m[i][t][st][p];
        int hi = t ? d[i] : b - 1; ll res = 0;
        for ( int x = 0; x <= hi; x++){
            if ( st && x<p) continue;
            int nt = (t && x == hi), ns = st || x, np=st || x? x:0;
            res = (res + dp(i+1, nt,ns,np,d,b,m)) % M;
        }
        return m[i][t][st][p] = res;
    }

    ll f(string s, int b){
        auto d = cv(s,b);
        vector m(d.size(), vector(2,vector(2, vector<ll>(b,-1))));
        return dp(0,1,0,0,d,b,m);
    }
    
    int countNumbers(string l, string r, int b) {
        return (f(r,b)- (l=="0" ? 0: f(ds(l),b))+M)%M;
    }
};
```

##
```py
class Solution:
    MOD = 10**9 + 7

    def cv(self, s, b):
        v = []

        while s != "0":
            r = 0
            ns = ""

            for c in s:
                cur = r * 10 + int(c)
                q = cur // b
                r = cur % b

                if ns or q:
                    ns += str(q)

            v.append(r)
            s = ns if ns else "0"

        v.reverse()
        return v if v else [0]

    def ds(self, s):
        s = list(s)

        for i in range(len(s) - 1, -1, -1):
            if s[i] != "0":
                s[i] = str(int(s[i]) - 1)
                break
            s[i] = "9"

        res = "".join(s).lstrip("0")
        return res if res else "0"

    def dp(self, i, tight, started, prev, digits, b, memo):
        if i == len(digits):
            return 1

        key = (i, tight, started, prev)

        if key in memo:
            return memo[key]

        hi = digits[i] if tight else b - 1
        res = 0

        for x in range(hi + 1):
            if started and x < prev:
                continue

            ntight = tight and x == hi
            nstarted = started or x != 0
            nprev = x if nstarted else 0

            res += self.dp(
                i + 1,
                ntight,
                nstarted,
                nprev,
                digits,
                b,
                memo
            )

            res %= self.MOD

        memo[key] = res
        return res

    def f(self, s, b):
        digits = self.cv(s, b)
        memo = {}

        return self.dp(
            0,
            True,
            False,
            0,
            digits,
            b,
            memo
        )

    def countNumbers(self, l, r, b):
        right = self.f(r, b)

        if l == "0":
            left = 0
        else:
            left = self.f(self.ds(l), b)

        return (right - left + self.MOD) % self.MOD
```
