## Brute Force Approach
```
class Solution {
    public boolean isValidSudoku(char[][] board) {
        
        // Check all rows
        for (int r = 0; r < 9; r++) {
            boolean[] seen = new boolean[9];
            for (int c = 0; c < 9; c++) {
                if (board[r][c] == '.') continue;
                int num = board[r][c] - '1';
                if (seen[num]) return false;
                seen[num] = true;
            }
        }

        // Check all columns
        for (int c = 0; c < 9; c++) {
            boolean[] seen = new boolean[9];
            for (int r = 0; r < 9; r++) {
                if (board[r][c] == '.') continue;
                int num = board[r][c] - '1';
                if (seen[num]) return false;
                seen[num] = true;
            }
        }

        // Check all 3×3 boxes
        for (int box = 0; box < 9; box++) {
            boolean[] seen = new boolean[9];
            int br = (box / 3) * 3;
            int bc = (box % 3) * 3;
            for (int r = br; r < br + 3; r++) {
                for (int c = bc; c < bc + 3; c++) {
                    if (board[r][c] == '.') continue;
                    int num = board[r][c] - '1';
                    if (seen[num]) return false;
                    seen[num] = true;
                }
            }
        }
        return true;
    }
}
```

## Hashing / Frequency-Counting Approach

```
class Solution {
    public boolean isValidSudoku(char[][] board) {
        int[][] rows = new int[9][9];
        int[][] cols = new int[9][9];
        int[][] boxes = new int[9][9];
        for (int r = 0; r < 9; r++) {
            for (int c = 0; c < 9; c++) {
                if (board[r][c] == '.') continue;
                int num = board[r][c] - '1';
                int boxIndex = (r / 3) * 3 + (c / 3);
                if (rows[r][num] == 1 || cols[c][num] == 1 || boxes[boxIndex][num] == 1) return false;
                rows[r][num] = 1;
                cols[c][num] = 1;
                boxes[boxIndex][num] = 1;
            }
        }
        return true;
    }
}
```

## HashSet Encoding Approach
```
class Solution {
    public boolean isValidSudoku(char[][] board) {
         Set<String> seen = new HashSet<>();
        for (int r = 0; r < 9; r++) {
            for (int c = 0; c < 9; c++) {
                char val = board[r][c];
                if (val == '.') continue;
                if (!seen.add(val + " in row " + r) ||
                    !seen.add(val + " in col " + c) ||
                    !seen.add(val + " in box " + (r/3) + "-" + (c/3))) {
                    return false;
                }
            }
        }
        return true;
    }
}
```

## Bitmasking Approach
```
class Solution {
    public boolean isValidSudoku(char[][] board) {
        int[] rows = new int[9];
        int[] cols = new int[9];
        int[] boxes = new int[9];
        for (int r = 0; r < 9; r++) {
            for (int c = 0; c < 9; c++) {
                if (board[r][c] == '.') continue;
                int bit = 1 << (board[r][c] - '1');
                int box = (r / 3) * 3 + (c / 3);
                if ((rows[r] & bit) != 0 || 
                    (cols[c] & bit) != 0 || 
                    (boxes[box] & bit) != 0) return false;
                rows[r] |= bit;
                cols[c] |= bit;
                boxes[box] |= bit;
            }
        }
        return true;
    }
}
```

## Boolean Array Approach
```
class Solution {
    public boolean isValidSudoku(char[][] board) {
        boolean[][] rows = new boolean[9][9];
        boolean[][] cols = new boolean[9][9];
        boolean[][] boxes = new boolean[9][9];
        for (int r = 0; r < 9; r++) {
            for (int c = 0; c < 9; c++) {
                if (board[r][c] == '.') continue;
                int num = board[r][c] - '1';
                int box = (r / 3) * 3 + (c / 3);
                if (rows[r][num] || cols[c][num] || boxes[box][num]) return false;
                rows[r][num] = cols[c][num] = boxes[box][num] = true;
            }
        }
        return true;
    }
}
```

## One Pass Single Array
```
class Solution {
    public boolean isValidSudoku(char[][] board) {
        boolean[] seen = new boolean[400];
        for (int r = 0; r < 9; r++) {
            for (int c = 0; c < 9; c++) {
                if (board[r][c] == '.') continue;
                int num = board[r][c] - '1';
                int box = (r / 3) * 3 + (c / 3);

                int rowIndex = r * 9 + num;
                int colIndex = 100 + c * 9 + num;
                int boxIndex = 200 + box * 9 + num;

                if (seen[rowIndex] || seen[colIndex] || seen[boxIndex]) return false;

                seen[rowIndex] = seen[colIndex] = seen[boxIndex] = true;
            }
        }
        return true;
    }
}
```
