## Brute Force - TLE
```
class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);
        Set<List<Integer>> res = new HashSet<>();
        backtrack(candidates, target, 0, new ArrayList<>(), res);
        return new ArrayList<>(res);
    }

    private void backtrack(int[] candidates, int target, int start, List<Integer> temp, Set<List<Integer>> res) {
        if (sum(temp) == target) {
            res.add(new ArrayList<>(temp));
        }
        if (start >= candidates.length) return;

        for (int i = start; i < candidates.length; i++) {
            temp.add(candidates[i]);
            backtrack(candidates, target, i + 1, temp, res);
            temp.remove(temp.size() - 1);
        }
    }

    private int sum(List<Integer> list) {
        int s = 0;
        for (int num : list) s += num;
        return s;
    }
}
```

## DP with HashSet
```
class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);
        
        List<Set<List<Integer>>> dp = new ArrayList<>();
        for (int i = 0; i <= target; i++) dp.add(new HashSet<>());
        dp.get(0).add(new ArrayList<>());
        
        for (int num : candidates) {
            for (int t = target; t >= num; t--) {
                for (List<Integer> prev : dp.get(t - num)) {
                    List<Integer> newComb = new ArrayList<>(prev);
                    newComb.add(num);
                    dp.get(t).add(newComb);
                }
            }
        }
        return new ArrayList<>(dp.get(target));
    }
}
```

## Bitmask Enumeration (Subsets Method)
```
class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);
        Set<List<Integer>> res = new HashSet<>();
        int n = candidates.length;
        for (int mask = 0; mask < (1 << n); mask++) {
            List<Integer> subset = new ArrayList<>();
            int sum = 0;
            for (int i = 0; i < n; i++) {
                if ((mask & (1 << i)) != 0) {
                    sum += candidates[i];
                    subset.add(candidates[i]);
                }
            }
            if (sum == target) res.add(subset);
        }
        return new ArrayList<>(res);
    }
}
```

## divide and conquer -TLE
```
class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);
        int n = candidates.length, mid = n / 2;

        List<List<Integer>> left = subsets(Arrays.copyOfRange(candidates, 0, mid));
        List<List<Integer>> right = subsets(Arrays.copyOfRange(candidates, mid, n));

        Map<Integer, List<List<Integer>>> map = new HashMap<>();
        for (List<Integer> r : right) {
            int sum = r.stream().mapToInt(Integer::intValue).sum();
            map.computeIfAbsent(sum, k -> new ArrayList<>()).add(r);
        }

        Set<List<Integer>> res = new HashSet<>();
        for (List<Integer> l : left) {
            int sum = l.stream().mapToInt(Integer::intValue).sum();
            int need = target - sum;
            if (map.containsKey(need)) {
                for (List<Integer> r : map.get(need)) {
                    List<Integer> combo = new ArrayList<>(l);
                    combo.addAll(r);
                    Collections.sort(combo);
                    res.add(combo);
                }
            }
        }
        return new ArrayList<>(res);
    }

    private List<List<Integer>> subsets(int[] arr) {
        List<List<Integer>> res = new ArrayList<>();
        res.add(new ArrayList<>());
        for (int num : arr) {
            List<List<Integer>> newSubs = new ArrayList<>();
            for (List<Integer> s : res) {
                List<Integer> t = new ArrayList<>(s);
                t.add(num);
                newSubs.add(t);
            }
            res.addAll(newSubs);
        }
        return res;
    }
}
```

## classic backtracking with duplicate skipping
```
class Solution {
    private List<List<Integer>> res = new ArrayList<>();
    private List<Integer> path = new ArrayList<>();
    private int[] nums;


    public List<List<Integer>> combinationSum2(int[] candidates, int target) {
        Arrays.sort(candidates);
        nums = candidates;
        dfs(0, target);
        return res;
    }

    private void dfs(int idx, int remain) {
        if (remain == 0) {
            res.add(new ArrayList<>(path));
            return;
        }
        if (idx >= nums.length || remain < nums[idx]) return;

        for (int i = idx; i < nums.length; i++) {
            if (i > idx && nums[i] == nums[i - 1]) continue;
            path.add(nums[i]);
            dfs(i + 1, remain - nums[i]);
            path.remove(path.size() - 1);
        }
    }
}
```
