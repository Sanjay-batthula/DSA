## Brute Force
```
class Solution {
    public boolean repeatedSubstringPattern(String s) {
        int n = s.length();
        for (int len = 1; len <= n / 2; len++) {
            if (n % len == 0) {
                String sub = s.substring(0, len);
                StringBuilder sb = new StringBuilder();
                int times = n / len;
                for (int i = 0; i < times; i++) sb.append(sub);
                if (sb.toString().equals(s)) return true;
            }
        }
        return false;
    }
}
```


## string doubling/rotation trick
```
class Solution {
    public boolean repeatedSubstringPattern(String s) {
        String str = s + s;
        return str.substring(1, str.length() - 1).contains(s);
    }
}
```

## KMP approach
```
class Solution {
    public boolean repeatedSubstringPattern(String s) {
        int n = s.length();
        int[] lps = new int[n];
        int len = 0, i = 1;
        while (i < n) {
            if (s.charAt(i) == s.charAt(len)) {
                lps[i++] = ++len;
            } else {
                if (len > 0) len = lps[len - 1];
                else lps[i++] = 0;
            }
        }
        int l = lps[n - 1];
        return l > 0 && n % (n - l) == 0;
    }
}
```

## Divide & Check approach
```
class Solution {
    public boolean repeatedSubstringPattern(String s) {
        int n = s.length();
        for (int len = 1; len <= n / 2; len++) {
            if (n % len == 0) {
                if (check(s, len)) {
                    return true;
                }
            }
        }
        return false;
    }

    private boolean check(String s, int len) {
        String sub = s.substring(0, len);
        for (int i = len; i < s.length(); i += len) {
            if (!s.startsWith(sub, i)) {
                return false; 
            }
        }
        return true;
    }
}
```
