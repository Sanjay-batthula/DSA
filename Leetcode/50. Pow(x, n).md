## Brute force - Time Limit Exceeded

```
class Solution {
    public double myPow(double x, int n) {
        long k = n;
        if (k < 0) k = -k;
        double ans = 1;
        for (long i = 0; i < k; i++) ans *= x;
        return n >= 0 ? ans : 1 / ans;
    }
}
```

## binary exponentiation
```
class Solution {
    public double myPow(double x, int n) {
        return n >= 0 ? qpow(x, n) : 1 / qpow(x, -(long) n);
    }

    private double qpow(double a, long n) {
        double ans = 1;
        for (; n > 0; n >>= 1) {
            if ((n & 1) == 1) {
                ans = ans * a;
            }
            a = a * a;
        }
        return ans;
    }
}
```


## Recursion
```
class Solution {
    public double myPow(double x, int n) {
        long k = n;
        return k >= 0 ? solve(x, k) : 1 / solve(x, -k);
    }

    private double solve(double x, long n) {
        if (n == 0) return 1;
        double half = solve(x, n / 2);
        if (n % 2 == 0) return half * half;
        return half * half * x;
    }
}
```

## Bitwise Fast Power
```
class Solution {
    public double myPow(double x, int n) {
        long p = n;
        if (p < 0) {
            x = 1 / x;
            p = -p;
        }
        double ans = 1;
        while (p > 0) {
            if ((p & 1) == 1) ans *= x;
            x *= x;
            p >>= 1;
        }
        return ans;
    }
}
```

## Branchless Multiplication
```
class Solution {
    public double myPow(double x, int n) {
        long p = n;
        if (p < 0) {
            x = 1 / x;
            p = -p;
        }
        double ans = 1;
        while (p > 0) {
            ans *= ((p & 1) == 1) ? x : 1;
            x *= x;
            p >>= 1;
        }
        return ans;
    }
}
```

## Iterative Splitting
```
class Solution {
    public double myPow(double x, int n) {
        long p = n;
        if (p < 0) {
            x = 1 / x;
            p = -p;
        }
        double ans = 1;
        double curr = x;

        while (p > 0) {
            if ((p & 1) == 1) ans *= curr;
            curr *= curr;
            p >>= 1;
        }
        return ans;
    }
}
```
