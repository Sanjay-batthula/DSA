## Brute Force - TLE
```
class Solution {
    public int maxSubArray(int[] nums) {
        int max=Integer.MIN_VALUE;
        for(int i=0;i<nums.length;i++){
            int sum=0;
            for(int j=i;j<nums.length;j++){
                sum+=nums[j];
                max=Math.max(max,sum);
            }
        }
        return max;
    }
}
```

## Kadane’s Algo
```
class Solution {
    public int maxSubArray(int[] nums) {
        int maxSum=nums[0];
        int currSum=nums[0];
        for (int i=1;i<nums.length; i++) {
            currSum = Math.max(nums[i], currSum + nums[i]);
            maxSum = Math.max(maxSum, currSum);
        }
        return maxSum;
    }
}
```

## Divide & Conquer
```
class Solution {
    public int maxSubArray(int[] nums) {
        return divide(nums, 0, nums.length - 1);
    }
    private int divide(int[] a, int l, int r) {
        if (l == r) return a[l];
        int mid = (l + r) / 2;
        int left = divide(a, l, mid);
        int right = divide(a, mid + 1, r);
        int cross = crossSum(a, l, mid, r);
        return Math.max(Math.max(left, right), cross);
    }
    private int crossSum(int[] a, int l, int mid, int r) {
        int leftMax = Integer.MIN_VALUE, sum = 0;
        for (int i = mid; i >= l; i--) {
            sum += a[i];
            leftMax = Math.max(leftMax, sum);
        }
        int rightMax = Integer.MIN_VALUE; sum = 0;
        for (int i = mid + 1; i <= r; i++) {
            sum += a[i];
            rightMax = Math.max(rightMax, sum);
        }
        return leftMax + rightMax;
    }
}
```

## Prefix Sum Removal Trick
```
class Solution {
    public int maxSubArray(int[] nums) {
        int prefix = 0, minPrefix = 0, max = Integer.MIN_VALUE;
        for (int x : nums) {
            prefix += x;
            max = Math.max(max, prefix - minPrefix);
            minPrefix = Math.min(minPrefix, prefix);
        }
        return max;
    }
}
```


## Dynamic Programming
```
class Solution {
    public int maxSubArray(int[] nums) {
        int dp = nums[0], max = nums[0];
        for(int i=1;i<nums.length;i++){
            dp = Math.max(nums[i], dp + nums[i]);
            max = Math.max(max, dp);
        }
        return max;
    }
}
```
