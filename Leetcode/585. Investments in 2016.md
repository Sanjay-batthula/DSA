# 585. Investments in 2016

## Subquery Filtering (IN + GROUP BY + HAVING)
```
# Write your MySQL query statement below
SELECT ROUND(SUM(TIV_2016), 2) AS TIV_2016 
FROM insurance
WHERE TIV_2015 IN (
    SELECT TIV_2015
    FROM insurance 
    GROUP BY TIV_2015
    HAVING COUNT(*) > 1
)
AND CONCAT(LAT, LON) IN (
    SELECT CONCAT(LAT, LON)
    FROM insurance
    GROUP BY LAT, LON
    HAVING COUNT(*) = 1
);
```

## CTE + INNER JOIN (Relational Join-based Filtering)
```
# Write your MySQL query statement below
WITH tiv_count AS (
    SELECT tiv_2015
    FROM insurance
    GROUP BY tiv_2015
    HAVING COUNT(*) > 1
),
loc_unique AS (
    SELECT lat, lon
    FROM insurance
    GROUP BY lat, lon
    HAVING COUNT(*) = 1
)
SELECT ROUND(SUM(i.tiv_2016), 2) AS tiv_2016
FROM insurance i
JOIN tiv_count t ON i.tiv_2015 = t.tiv_2015
JOIN loc_unique l ON i.lat = l.lat AND i.lon = l.lon;
```
