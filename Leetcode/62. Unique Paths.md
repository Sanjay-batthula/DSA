## Brute Force - TLE
```
class Solution {
    public int uniquePaths(int m, int n) {
        return dfs(0, 0, m, n);
    }

    int dfs(int i, int j, int m, int n) {
        if(i == m - 1 && j == n - 1) return 1;
        if(i >= m || j >= n) return 0;
        return dfs(i + 1, j, m, n) + dfs(i, j + 1, m, n);
    }
}
```

## Dynamic Programming (Tabulation)
```
class Solution {
    public int uniquePaths(int m, int n) {
        var f = new int[m][n];
        f[0][0] = 1;
        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                if (i > 0) {
                    f[i][j] += f[i - 1][j];
                }
                if (j > 0) {
                    f[i][j] += f[i][j - 1];
                }
            }
        }
        return f[m - 1][n - 1];
    }
}
```

## DP — Space Optimized (1D DP)
```
class Solution {
    public int uniquePaths(int m, int n) {
        int[] dp = new int[n];
        dp[0] = 1;
        for(int i = 0; i < m; i++) {
            for(int j = 1; j < n; j++) {
                dp[j] += dp[j - 1];
            }
        }
        return dp[n - 1];
    }
}
```

## Combinatorics (Most OP Approach)
```
class Solution {
    public int uniquePaths(int m, int n) {
        long res = 1;
        int k = Math.min(m - 1, n - 1);
        for(int i = 1; i <= k; i++){
            res = res * (m + n - 2 - (k - i)) / i;
        }
        return (int)res;
    }
}

```

## Memoization (Top-Down DP)
```
class Solution {
    public int uniquePaths(int m, int n) {
        int[][] dp = new int[m][n];
        return solve(0, 0, m, n, dp);
    }

    int solve(int i, int j, int m, int n, int[][] dp) {
        if(i == m - 1 && j == n - 1) return 1;
        if(i >= m || j >= n) return 0;
        if(dp[i][j] != 0) return dp[i][j];
        return dp[i][j] = solve(i + 1, j, m, n, dp) + solve(i, j + 1, m, n, dp);
    }
}
```
