# 626. Exchange Seats

## Conditional ID Remapping
```
SELECT ( CASE
            WHEN id%2 != 0 AND id != counts THEN id+1
            WHEN id%2 != 0 AND id = counts THEN id
            ELSE id-1
        END) AS id, student
FROM seat, (select count(*) as counts from seat) 
AS seat_counts
ORDER BY id ASC
```

## Self Join for Adjacent Row Swapping
```
SELECT
    s1.id,
    COALESCE(
        CASE
            WHEN s1.id % 2 = 1 THEN s2.student
            ELSE s3.student
        END,
        s1.student
    ) AS student
FROM Seat s1
LEFT JOIN Seat s2
    ON s1.id % 2 = 1
    AND s1.id + 1 = s2.id
LEFT JOIN Seat s3
    ON s1.id % 2 = 0
    AND s1.id - 1 = s3.id
ORDER BY s1.id;
```

## Window Function Based Adjacent Swapping
```
WITH cte AS (
    SELECT
        id,
        student,
        LEAD(student) OVER(ORDER BY id) AS next_student,
        LAG(student) OVER(ORDER BY id) AS prev_student,
        COUNT(*) OVER() AS cnt
    FROM Seat
)

SELECT
    id,
    CASE
        WHEN id % 2 = 1 AND id != cnt THEN next_student
        WHEN id % 2 = 0 THEN prev_student
        ELSE student
    END AS student
FROM cte
ORDER BY id;
```
