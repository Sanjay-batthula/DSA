## Brute Force
```
class Solution {
    public boolean isMonotonic(int[] nums) {
        boolean increasing = true;
        boolean decreasing = true;
        
        for (int i = 1; i < nums.length; i++) {
            if (nums[i] < nums[i - 1]) increasing = false;
            if (nums[i] > nums[i - 1]) decreasing = false;
        }
        
        return increasing || decreasing;
    }
}
```


## greedy / single-pass approach
```
class Solution {
    public boolean isMonotonic(int[] nums) {
        boolean asc = false, desc = false;
        for (int i = 1; i < nums.length; ++i) {
            if (nums[i - 1] < nums[i]) {
                asc = true;
            } else if (nums[i - 1] > nums[i]) {
                desc = true;
            }
            if (asc && desc) {
                return false;
            }
        }
        return true;
    }
}
```

## Using Sorting 
```
class Solution {
    public boolean isMonotonic(int[] nums) {
        int[] asc = nums.clone();
        int[] desc = nums.clone();
        Arrays.sort(asc);
        Arrays.sort(desc);
        reverse(desc);
        return Arrays.equals(nums, asc) || Arrays.equals(nums, desc);
    }

    private void reverse(int[] arr) {
        int i = 0, j = arr.length - 1;
        while (i < j) {
            int tmp = arr[i];
            arr[i] = arr[j];
            arr[j] = tmp;
            i++;
            j--;
        }
    }
}

```


## Sign of Differences
```
class Solution {
    public boolean isMonotonic(int[] nums) {
        int direction = 0;
        for (int i = 1; i < nums.length; i++) {
            int diff = nums[i] - nums[i - 1];
            if (diff == 0) continue;
            if (direction == 0) direction = diff > 0 ? 1 : -1;
            else if ((direction > 0 && diff < 0) || (direction < 0 && diff > 0)) return false;
        }
        return true;
    }
}
```

## Two pointer Approach
```
class Solution {
    public boolean isMonotonic(int[] nums) {
        int i = 0, j = 1;
        boolean inc = false, dec = false;
        while (j < nums.length) {
            if (nums[j] > nums[i]) inc = true;
            else if (nums[j] < nums[i]) dec = true;
            if (inc && dec) return false;
            i++;
            j++;
        }
        return true;
    }
}
```
