
## Brute force
```
class Solution {
    public boolean isPalindrome(int x) {
        String s = String.valueOf(x);
        String rev = new StringBuilder(s).reverse().toString();
        return s.equals(rev);
    }
}
```


## Half Reversal Approach
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0 || (x % 10 == 0 && x != 0)) return false;
        int rev = 0;
        while (x > rev) {
            rev = rev * 10 + x % 10;
            x /= 10;
        }
        return x == rev || x == rev / 10;
    }
}
```


## Full Reversal 
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        long rev = 0, n = x;
        while (n != 0) {
            rev = rev * 10 + n % 10;
            n /= 10;
        }
        return rev == x;
    }
}
```


## Half Reversal (Loop Variant)
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0 || (x > 0 && x % 10 == 0)) {
            return false;
        }
        int y = 0;
        for (; y < x; x /= 10) {
            y = y * 10 + x % 10;
        }
        return x == y || x == y / 10;
    }
}
```

## Array Conversion Approach - Runtime Error
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        int n = (int) Math.log10(x) + 1;
        int[] digits = new int[n];
        for (int i = 0; i < n; i++) {
            digits[i] = x % 10;

        }
        for (int i = 0; i < n / 2; i++) {
            if (digits[i] != digits[n - i - 1]) return false;
        }
        return true;
    }

## Two-Pointer String Approach
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        String s = String.valueOf(x);
        int left = 0, right = s.length() - 1;
        while (left < right) {
            if (s.charAt(left++) != s.charAt(right--)) return false;
        }
        return true;
    }
}
```

## Digit Comparison Without Reversal
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        int div = 1;
        while (x / div >= 10) div *= 10;
        while (x != 0) {
            int left = x / div;
            int right = x % 10;
            if (left != right) return false;
            x = (x % div) / 10;
            div /= 100;
        }
        return true;
    }
}
```

## Recursive Numeric Check
```
class Solution {
    private int div = 1;
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        while (x / div >= 10) div *= 10;
        return helper(x);
    }
    private boolean helper(int x) {
        if (x == 0) return true;
        int left = x / div;
        int right = x % 10;
        if (left != right) return false;
        x = (x % div) / 10;
        div /= 100;
        return helper(x);
    }
}
```

## Mathematical Mirror Check (Using Logarithms)
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        int digits = (int)Math.log10(x) + 1;
        for (int i = 0; i < digits / 2; i++) {
            int left = (x / (int)Math.pow(10, digits - i - 1)) % 10;
            int right = (x / (int)Math.pow(10, i)) % 10;
            if (left != right) return false;
        }
        return true;
    }
}
```

## Stream / Functional Approach
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        List<Integer> digits = new ArrayList<>();
        int temp = x;
        while (temp != 0) {
            digits.add(temp % 10);
            temp /= 10;
        }
        return IntStream.range(0, digits.size() / 2)
                .allMatch(i -> Objects.equals(digits.get(i), digits.get(digits.size() - i - 1)));
    }
}
```

## Recursive Two-Pointer (String version)
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        String s = String.valueOf(x);
        return helper(s, 0, s.length() - 1);
    }
    private boolean helper(String s, int left, int right) {
        if (left >= right) return true;
        return s.charAt(left) == s.charAt(right) && helper(s, left + 1, right - 1);
    }
}
```

## Queue + Stack Comparison
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        String s = String.valueOf(x);
        Stack<Character> stack = new Stack<>();
        Queue<Character> queue = new LinkedList<>();
        for (char c : s.toCharArray()) {
            stack.push(c);
            queue.add(c);
        }
        while (!stack.isEmpty()) {
            if (!stack.pop().equals(queue.poll())) return false;
        }
        return true;
    }
}
```

## Pattern Matching via Regex
```
class Solution {
    public boolean isPalindrome(int x) {
        if (x < 0) return false;
        String s = String.valueOf(x);
        return s.matches(new StringBuilder(s).reverse().toString());
    }
}
