## Brute Force
```
class Solution {
    public int findLucky(int[] arr) {
        int ans = -1;
        for (int i = 0; i < arr.length; i++) {
            int count = 0;
            for (int j = 0; j < arr.length; j++) {
                if (arr[i] == arr[j]) count++;
            }
            if (arr[i] == count) ans = Math.max(ans, arr[i]);
        }
        return ans;
    }
}
```


## Frequency Array / Counting Approach
```
class Solution {
    public int findLucky(int[] arr) {
        int[] cnt=new int[501];
        for(int x:arr)++cnt[x];
        for(int x=500;x>0;--x)if(x==cnt[x])return x;
        return -1;
    }
}
```

## HashMap Approach
```
class Solution {
    public int findLucky(int[] arr) {
        Map<Integer,Integer> map = new HashMap<>();
        for(int x : arr)
            map.put(x, map.getOrDefault(x, 0) + 1);
        
        int ans = -1;
        for(int key : map.keySet())
            if(key == map.get(key))
                ans = Math.max(ans, key);
        
        return ans;
    }
}
```

## Sorting + Frequency Count
```
class Solution {
    public int findLucky(int[] arr) {
        Arrays.sort(arr);
        int ans = -1;
        int i = 0;
        while (i < arr.length) {
            int num = arr[i];
            int count = 0;
            while (i < arr.length && arr[i] == num) {
                count++;
                i++;
            }
            if (num == count) ans = num;
        }
        return ans;
    }
}
```
