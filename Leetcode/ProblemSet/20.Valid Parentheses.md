## Brute Force
```
class Solution {
    public boolean isValid(String s) {
        while (s.contains("()") || s.contains("{}") || s.contains("[]")) {
            s = s.replace("()", "").replace("{}", "").replace("[]", "");
        }
        return s.isEmpty();
    }
}
```


## Stack-based approach
```
class Solution {
    public boolean isValid(String s) {
        char[] arr = new char[s.length()];
        int top = -1;
        for (char c : s.toCharArray()) {
            if (c == '(' || c == '[' || c == '{') {
                arr[++top] = c;
            } else {
                if (top == -1) return false;
                if (c == ')' && arr[top] != '(') return false;
                if (c == ']' && arr[top] != '[') return false;
                if (c == '}' && arr[top] != '{') return false;
                top--;
            }
        }
        return top == -1;
    }
}
```

## Deque Stack Approach
```
class Solution {
    public boolean isValid(String s) {
        Deque<Character> stk = new ArrayDeque<>();
        for (char c : s.toCharArray()) {
            if (c == '(' || c == '{' || c == '[') {
                stk.push(c);
            } else if (stk.isEmpty() || !match(stk.pop(), c)) {
                return false;
            }
        }
        return stk.isEmpty();
    }
    private boolean match(char l, char r) {
        return (l == '(' && r == ')') || (l == '{' && r == '}') || (l == '[' && r == ']');
    }
}
```

## Expected Bracket Stack Approach
```
class Solution {
    public boolean isValid(String s) {
        Stack<Character> stack = new Stack<>();
        for (char c : s.toCharArray()) {
            if (c == '(') stack.push(')');
            else if (c == '[') stack.push(']');
            else if (c == '{') stack.push('}');
            else {
                if (stack.isEmpty() || stack.pop() != c) return false;
            }
        }
        return stack.isEmpty();
    }
}
```


## Recursive Approach
```
class Solution {
    public boolean isValid(String s) {
        if (s.isEmpty()) return true;
        String replaced = s.replace("()", "").replace("{}", "").replace("[]", "");
        if (replaced.equals(s)) return false;
        return isValid(replaced);
    }
}
```

## Using a HashMap for Pair Mapping
```
class Solution {
    public boolean isValid(String s) {
        Map<Character, Character> map = Map.of(')', '(', '}', '{', ']', '[');
        Stack<Character> stack = new Stack<>();
        for (char c : s.toCharArray()) {
            if (map.containsValue(c)) stack.push(c);
            else if (stack.isEmpty() || stack.pop() != map.get(c)) return false;
        }
        return stack.isEmpty();
    }
}

## String Builder Stack Simulation
```
class Solution {
    public boolean isValid(String s) {
        StringBuilder sb = new StringBuilder();
        for (char c : s.toCharArray()) {
            int len = sb.length();
            if (c == ')' && len > 0 && sb.charAt(len - 1) == '(') sb.deleteCharAt(len - 1);
            else if (c == '}' && len > 0 && sb.charAt(len - 1) == '{') sb.deleteCharAt(len - 1);
            else if (c == ']' && len > 0 && sb.charAt(len - 1) == '[') sb.deleteCharAt(len - 1);
            else sb.append(c);
        }
        return sb.length() == 0;
    }
}
