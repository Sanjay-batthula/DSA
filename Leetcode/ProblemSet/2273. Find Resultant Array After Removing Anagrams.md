## Brute Force
```
class Solution {
    public List<String> removeAnagrams(String[] words) {
        List<String> res = new ArrayList<>();
        res.add(words[0]);
        for (int i = 1; i < words.length; i++) {
            char[] a = words[i].toCharArray();
            char[] b = words[i - 1].toCharArray();
            Arrays.sort(a);
            Arrays.sort(b);
            if (!Arrays.equals(a, b)) res.add(words[i]);
        }
        return res;
    }
}
```


## Single Pass Optimized Approach
```
class Solution {
    public List<String> removeAnagrams(String[] words) {
        List<String> res = new ArrayList<>();
        String prevSorted = "";
        for (String w : words) {
            char[] ch = w.toCharArray();
            Arrays.sort(ch);
            String sorted = new String(ch);
            if (!sorted.equals(prevSorted)) {
                res.add(w);
                prevSorted = sorted;
            }
        }
        return res;
    }
}
```

## Frequency Count
```
class Solution {
    public List<String> removeAnagrams(String[] words) {
        List<String> res = new ArrayList<>();
        String prevKey = "";
        for (String w : words) {
            int[] count = new int[26];
            for (char c : w.toCharArray()) count[c - 'a']++;
            String key = Arrays.toString(count);
            if (!key.equals(prevKey)) {
                res.add(w);
                prevKey = key;
            }
        }
        return res;
    }
}
```

##  HashMap / Signature Cache Approach
```
class Solution {
    public List<String> removeAnagrams(String[] words) {
        List<String> res = new ArrayList<>();
        Map<String, String> cache = new HashMap<>();
        String prev = "";
        for (String w : words) {
            char[] ch = w.toCharArray();
            Arrays.sort(ch);
            String key = new String(ch);
            cache.putIfAbsent(w, key);
            if (!cache.get(w).equals(prev)) {
                res.add(w);
                prev = cache.get(w);
            }
        }
        return res;
    }
}
```
