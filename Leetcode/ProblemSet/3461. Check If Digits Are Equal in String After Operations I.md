## Brute Force
```
class Solution {
    public boolean hasSameDigits(String s) {
        List<Integer> digits = new ArrayList<>();
        for (char c : s.toCharArray()) {
            digits.add(c - '0');
        }

        while (digits.size() > 2) {
            List<Integer> next = new ArrayList<>();
            for (int i = 0; i < digits.size() - 1; i++) {
                next.add((digits.get(i) + digits.get(i + 1)) % 10);
            }
            digits = next;
        }

        return digits.get(0).equals(digits.get(1));
    }
}
```

## combinatorial mathematics–based approach
```
class Solution {
    public boolean hasSameDigits(String s) {
        int n = s.length();
        int[] digits = new int[n];
        for (int i = 0; i < n; i++)
            digits[i] = s.charAt(i) - '0';
        int[] comb = new int[n];
        comb[0] = 1;
        for (int i = 1; i < n - 1; i++) {
            for (int j = i; j > 0; j--) {
                comb[j] = (comb[j] + comb[j - 1]) % 10;
            }
        }
        int left = 0, right = 0;
        for (int i = 0; i < n - 1; i++) {
            left = (left + comb[i] * digits[i]) % 10;
            right = (right + comb[i] * digits[i + 1]) % 10;
        }
        return left == right;
    }
}
```

## Dynamic Programming
```
class Solution {
    public boolean hasSameDigits(String s) {
        int n = s.length();
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) arr[i] = s.charAt(i) - '0';
        
        while (n > 2) {
            for (int i = 0; i < n - 1; i++) {
                arr[i] = (arr[i] + arr[i + 1]) % 10;
            }
            n--;
        }        
        return arr[0] == arr[1];
    }
}
```

## Recursion
```
class Solution {
    public boolean hasSameDigits(String s) {
        if (s.length() == 2) return s.charAt(0) == s.charAt(1);
        StringBuilder next = new StringBuilder();
        for (int i = 0; i < s.length() - 1; i++) {
            int val = (s.charAt(i) - '0' + s.charAt(i + 1) - '0') % 10;
            next.append(val);
        }
        return hasSameDigits(next.toString());
    }
}
```
