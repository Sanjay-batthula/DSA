class Solution {
    public char findTheDifference(String s, String t) {
        StringBuilder sb = new StringBuilder(s);
        for (int i = 0; i < t.length(); i++) {
            char c = t.charAt(i);
            int idx = sb.indexOf(String.valueOf(c));
            if (idx == -1) return c;
            sb.deleteCharAt(idx);
        }
        return ' ';
    }
}
