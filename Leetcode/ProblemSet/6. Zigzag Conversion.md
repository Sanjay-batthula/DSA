
## brute force
class Solution {
    public String convert(String s, int numRows) {
        if (numRows == 1 || s.length() <= numRows) return s;

        int n = s.length();
        char[][] grid = new char[numRows][n];
        for (int i = 0; i < numRows; i++) {
            for (int j = 0; j < n; j++) {
                grid[i][j] = ' ';
            }
        }

        int row = 0, col = 0;
        boolean goingDown = true;

        for (char c : s.toCharArray()) {
            grid[row][col] = c;
            if (goingDown) {
                if (row == numRows - 1) {
                    goingDown = false;
                    row--;
                    col++;
                } else {
                    row++;
                }
            } else {
                if (row == 0) {
                    goingDown = true;
                    row++;
                } else {
                    row--;
                    col++;
                }
            }
        }

        StringBuilder result = new StringBuilder();
        for (int i = 0; i < numRows; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] != ' ') {
                    result.append(grid[i][j]);
                }
            }
        }
        return result.toString();
    }
}



## Math Formula Approach
class Solution {
    public String convert(String s, int numRows) {
        if (numRows == 1 || s.length() <= numRows) return s;

        int n = s.length();
        int cycle = 2 * numRows - 2;
        StringBuilder result = new StringBuilder();

        for (int r = 0; r < numRows; r++) {
            for (int i = r; i < n; i += cycle) {
                result.append(s.charAt(i));
                if (r != 0 && r != numRows - 1 && i + cycle - 2 * r < n) {
                    result.append(s.charAt(i + cycle - 2 * r));
                }
            }
        }
        return result.toString();
    }
}
