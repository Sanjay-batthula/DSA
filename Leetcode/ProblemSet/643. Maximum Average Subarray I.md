## Brute Force
```
class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int sum = 0;
        for (int i = 0; i < k; i++) {
            sum += nums[i];
        }
        double maxAvg = (double) sum / k;
        
        for (int i = 1; i <= nums.length - k; i++) {
            sum = 0;
            for (int j = i; j < i + k; j++) {
                sum += nums[j];
            }
            double avg = (double) sum / k;
            maxAvg = Math.max(maxAvg, avg);
        }
        return maxAvg;
    }
}

## sliding window approach
```
class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int sum = 0;
        for (int i = 0; i < k; i++) {
            sum += nums[i];
        }
        double maxAvg = (double) sum / k;
        
        for (int i = k; i < nums.length; i++) {
            sum += nums[i] - nums[i - k];
            double avg = (double) sum / k;
            maxAvg = Math.max(maxAvg, avg);
        }
        return maxAvg;
    }
}
```

## Prefix Sum Array
```
class Solution {
    public double findMaxAverage(int[] nums, int k) {
        int n = nums.length;
        int[] prefix = new int[n + 1];
        for (int i = 0; i < n; i++) {
            prefix[i + 1] = prefix[i] + nums[i];
        }
        
        double maxAvg = (double)(prefix[k] - prefix[0]) / k;
        for (int i = 1; i <= n - k; i++) {
            double avg = (double)(prefix[i + k] - prefix[i]) / k;
            maxAvg = Math.max(maxAvg, avg);
        }
        return maxAvg;
    }
}
