
## Brute Force
```
class Solution {
    public boolean lemonadeChange(int[] bills) {
        List<Integer> wallet = new ArrayList<>();
        
        for (int bill : bills) {
            if (bill == 5) {
                wallet.add(5);
            } else if (bill == 10) {
                if (wallet.contains(5)) {
                    wallet.remove(Integer.valueOf(5));
                    wallet.add(10);
                } else {
                    return false;
                }
            } else { // bill == 20
                if (wallet.contains(10) && wallet.contains(5)) {
                    wallet.remove(Integer.valueOf(10));
                    wallet.remove(Integer.valueOf(5));
                } else if (Collections.frequency(wallet, 5) >= 3) {
                    wallet.remove(Integer.valueOf(5));
                    wallet.remove(Integer.valueOf(5));
                    wallet.remove(Integer.valueOf(5));
                } else {
                    return false;
                }
            }
        }
        return true;
    }
}

## Greedy Approach
```
class Solution {
    public boolean lemonadeChange(int[] bills) {
        int five = 0, ten = 0;
        for (int v : bills) {
            switch (v) {
                case 5 -> ++five;
                case 10 -> {
                    ++ten;
                    --five;
                }
                case 20 -> {
                    if (ten > 0) {
                        --ten;
                        --five;
                    } else {
                        five -= 3;
                    }
                }
            }
            if (five < 0) {
                return false;
            }
        }
        return true;
    }
}

## HashMap Approach
```
class Solution {
    public boolean lemonadeChange(int[] bills) {
        Map<Integer, Integer> cash = new HashMap<>();
        cash.put(5, 0);
        cash.put(10, 0);
        
        for (int bill : bills) {
            if (bill == 5) {
                cash.put(5, cash.get(5) + 1);
            } else if (bill == 10) {
                if (cash.get(5) <= 0) return false;
                cash.put(5, cash.get(5) - 1);
                cash.put(10, cash.get(10) + 1);
            } else { // 20
                if (cash.get(10) > 0 && cash.get(5) > 0) {
                    cash.put(10, cash.get(10) - 1);
                    cash.put(5, cash.get(5) - 1);
                } else if (cash.get(5) >= 3) {
                    cash.put(5, cash.get(5) - 3);
                } else {
                    return false;
                }
            }
        }
        return true;
    }
}

## Priority Queue Approach - Wrong Answer
```
class Solution {
    public boolean lemonadeChange(int[] bills) {
        PriorityQueue<Integer> pq = new PriorityQueue<>();
        
        for (int bill : bills) {
            pq.add(bill);
            int change = bill - 5;
            List<Integer> temp = new ArrayList<>();
            
            while (change > 0 && !pq.isEmpty()) {
                int smallest = pq.peek();
                if (smallest <= change) {
                    change -= smallest;
                    pq.poll();
                    temp.add(smallest);
                } else break;
            }
            
            if (change > 0) return false;
            pq.addAll(temp);
        }
        return true;
    }
}
